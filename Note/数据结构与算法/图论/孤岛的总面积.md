主要就是怎么标记孤岛的问题，

```java
import java.util.*;
 
public class Main<p> {
    static int[][] hash = {{-1, 0}, {1, 0}, {0, 1}, {0, -1}}; // 上下左右
    static int res = 0;
    static int tempNum = 0;
    static boolean isIsolated = true;
    public static void dfs(int[][] grid, int x, int y, boolean[][] visited){
        int col = grid.length;
        int row = grid[0].length;
        tempNum++;
        if (x == 0 || y == 0 || x == col - 1 || y == row - 1) {
            isIsolated = false; // 如果接触边缘，标记
        }
        for (int[] temp : hash){
            int nx = x + temp[0];
            int ny = y + temp[1];
            if(nx < 0 || ny < 0 || nx >= col || ny >= row || visited[nx][ny] || grid[nx][ny] == 0){
                continue;
            } else {
                visited[nx][ny] = true;
                dfs(grid, nx, ny, visited);
            }
        }
    }
 
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int col = scanner.nextInt(); // 行
        int row = scanner.nextInt(); // 列
        int[][] grid = new int[col][row];
        // 初始化图
        for (int i = 0; i < col; i++) {
            for (int j = 0; j < row; j++) {
                grid[i][j] = scanner.nextInt();
            }
        }
        boolean[][] visited = new boolean[col][row];
        for (int i = 0; i < col; i++) {
            for (int j = 0; j < row; j++) {
                if (grid[i][j] == 1 && !visited[i][j]){
                    tempNum = 0;
                    isIsolated = true;
                    visited[i][j] = true;
                    dfs(grid, i, j, visited);
                    if(isIsolated){
                        res += tempNum;
                    }
                }
            }
        }
        System.out.println(res);
    }
}
```
