class Solution {
    public int numFriendRequests(int[] ages) {
        int length = ages.length;
        int res = 0;
        for(int i = 0; i < length; i++){
            for(int j = 0; j < length; j++){
                if(i == j){
                    continue;
                }
                if(!(ages[j] <= 0.5 * ages[i] + 7) && !(ages[j] > ages[i]) && !(ages[j] > 100 && ages[i] < 100)){
                    res++;
                }
            }
        }
            return res;
    }
}
